    def retry_delay(response, attempt)
      fallback = [@retry_max_delay, @retry_base_delay * (2.0 ** [attempt - 1, 20].min)].min
      value = response['retry-after-ms'] if response.respond_to?(:[])
      if value && value.match?(/\A\d+(?:\.\d+)?\z/)
        return [@retry_max_delay, value.to_f / 1000.0].min
      end
      value = response['Retry-After'] if response.respond_to?(:[])
      return fallback unless value
      if value.match?(/\A\d+(?:\.\d+)?\z/)
        return [@retry_max_delay, value.to_f].min
      end
      begin
        require 'time'
        [[Time.httpdate(value) - Time.now, 0].max, @retry_max_delay].min
      rescue ArgumentError
        fallback
      end
    end
    def check_cancellation
      raise KajiCancellationError, 'request cancelled' if @cancelled && @cancelled.call
    end
    def retry_pause(delay)
      deadline = Process.clock_gettime(Process::CLOCK_MONOTONIC) + delay
      loop do
        check_cancellation
        remaining = deadline - Process.clock_gettime(Process::CLOCK_MONOTONIC)
        break if remaining <= 0
        sleep [remaining, 0.05].min
      end
    end
    def execute_with_retry(handler, request, idempotency_header)
      method = request.method.upcase
      key = request[idempotency_header || 'Idempotency-Key']
      replay_safe = %w[GET HEAD OPTIONS PUT DELETE].include?(method) || (%w[POST PATCH].include?(method) && key && !key.strip.empty?)
      token = @token_provider && @token_provider.token(cancelled: @cancelled)
      request["Authorization"] = "Bearer #{token}" if token
      refreshed = false
      attempt = 0
      loop do
        check_cancellation
        attempt += 1
        begin
          response = handler.call(request)
        rescue IOError, EOFError, SystemCallError, Timeout::Error
          raise unless replay_safe && attempt < @max_attempts
          retry_pause([@retry_max_delay, @retry_base_delay * (2.0 ** [attempt - 1, 20].min)].min)
          next
        end
        if response.code.to_i == 401 && @token_provider && replay_safe && !refreshed
          refreshed = true
          token = @token_provider.token(rejected_token: token, cancelled: @cancelled)
          request["Authorization"] = "Bearer #{token}"
          attempt -= 1
          next
        end
        return response unless replay_safe && attempt < @max_attempts && [408, 429, 500, 502, 503, 504].include?(response.code.to_i)
        retry_pause(retry_delay(response, attempt))
      end
    end
