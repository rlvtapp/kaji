            var raw=new String(requestBody(request),StandardCharsets.ISO_8859_1);var chunks=raw.split(java.util.regex.Pattern.quote("--"+boundary),-1);check(chunks.length==12&&chunks[0].isEmpty()&&chunks[11].equals("--\r\n"),"all repeated parts framed");
            var parsed=new HashMap<String,List<byte[]>>();

        }
