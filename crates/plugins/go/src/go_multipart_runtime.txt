
// KajiMultipartBody explicitly selects multipart/form-data for upload operations.
// Parts are buffered once, so replay-safe requests reuse identical bytes/boundaries.
// Do not mutate the body while a request is executing.
type KajiMultipartBody struct { Parts []KajiMultipartPart; ContentType string }
type KajiMultipartPart struct {
    Name string
    Filename string
    ContentType string
    Data []byte
    Headers http.Header
    Nested *KajiMultipartBody
}
func (body *KajiMultipartBody) AddText(name, value string) {
    body.Parts = append(body.Parts, KajiMultipartPart{Name:name, ContentType:"text/plain; charset=utf-8", Data:[]byte(value)})
}
func (body *KajiMultipartBody) AddFile(name, filename, contentType string, data []byte) {
    body.Parts = append(body.Parts, KajiMultipartPart{Name:name, Filename:filename, ContentType:contentType, Data:append([]byte(nil),data...)})
}
func (body *KajiMultipartBody) AddJSON(name string, value any) error {
    data, err := json.Marshal(value)
    if err != nil { return fmt.Errorf("kaji: encode multipart JSON: %w",err) }
    body.Parts = append(body.Parts, KajiMultipartPart{Name:name, ContentType:"application/json", Data:data})
    return nil
}
type kajiMultipartBuffer struct { bytes.Buffer }
func (buffer *kajiMultipartBuffer) Write(data []byte) (int,error) {
    if len(data) > (64<<20)-buffer.Len() { return 0,fmt.Errorf("kaji: multipart body exceeds 64 MiB") }
    return buffer.Buffer.Write(data)
}
func (body *KajiMultipartBody) encode() ([]byte,string,error) {
    return body.encodeDepth(0)
}
func (body *KajiMultipartBody) encodeDepth(depth int) ([]byte,string,error) {
    if depth>16{return nil,"",fmt.Errorf("kaji: multipart nesting exceeds 16 levels")}
    if body == nil { return nil,"",fmt.Errorf("kaji: multipart body is required") }
    if len(body.Parts)>10000 { return nil,"",fmt.Errorf("kaji: multipart exceeds 10000 parts") }
    media:=body.ContentType;if media==""{media="multipart/form-data"};parsed,parameters,err:=mime.ParseMediaType(media);if err!=nil||!strings.HasPrefix(parsed,"multipart/"){return nil,"",fmt.Errorf("kaji: multipart content type required")};if _,present:=parameters["boundary"];present{return nil,"",fmt.Errorf("kaji: multipart boundary is generated")}
    buffer:=&kajiMultipartBuffer{}
    writer:=multipart.NewWriter(buffer)
    for _,part:=range body.Parts {
        if (parsed=="multipart/form-data" && part.Name=="") || len(part.Name)>4096 || len(part.Filename)>4096 || strings.ContainsAny(part.Name+part.Filename,"\r\n\x00") {
            return nil,"",fmt.Errorf("kaji: invalid multipart name or filename")
        }
        if len(part.Data)>64<<20 {return nil,"",fmt.Errorf("kaji: multipart body exceeds 64 MiB")}
        data:=part.Data
        contentType:=part.ContentType
        if part.Nested!=nil {var err error;data,contentType,err=part.Nested.encodeDepth(depth+1);if err!=nil{return nil,"",err}}
        if contentType=="" {if part.Filename!="" {contentType="application/octet-stream"} else {contentType="text/plain; charset=utf-8"}}
        if len(contentType)>1024 {return nil,"",fmt.Errorf("kaji: invalid multipart content type")}
        if _,_,err:=mime.ParseMediaType(contentType);err!=nil{return nil,"",fmt.Errorf("kaji: invalid multipart content type")}
        headers:=textproto.MIMEHeader{}
        dispositionParameters:=map[string]string{}
        if part.Name!="" {dispositionParameters["name"]=part.Name}
        if part.Filename!="" {dispositionParameters["filename"]=part.Filename}
        if parsed=="multipart/form-data" {headers.Set("Content-Disposition",mime.FormatMediaType("form-data",dispositionParameters))} else if part.Name!=""||part.Filename!="" {headers.Set("Content-Disposition",mime.FormatMediaType("attachment",dispositionParameters))}
        headers.Set("Content-Type",contentType)
        if len(part.Headers)>64{return nil,"",fmt.Errorf("kaji: too many multipart headers")}
        for name,values:=range part.Headers {
            if len(name)==0||len(name)>256||strings.EqualFold(name,"Content-Disposition")||strings.EqualFold(name,"Content-Type"){return nil,"",fmt.Errorf("kaji: invalid multipart header")}
            for _,character:=range name {if !strings.ContainsRune("!#$%&'*+-.^_`|~0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz",character){return nil,"",fmt.Errorf("kaji: invalid multipart header")}}
            if len(values)>64{return nil,"",fmt.Errorf("kaji: too many multipart header values")}
            for _,value:=range values {if len(value)>8192||strings.ContainsAny(value,"\r\n\x00"){return nil,"",fmt.Errorf("kaji: invalid multipart header")};headers.Add(name,value)}
        }
        target,err:=writer.CreatePart(headers)
        if err!=nil{return nil,"",err}
        if _,err:=target.Write(data);err!=nil{return nil,"",err}
    }
    if err:=writer.Close();err!=nil{return nil,"",err}
    parameters["boundary"]=writer.Boundary()
    return buffer.Bytes(),mime.FormatMediaType(parsed,parameters),nil
}
func encodeKajiBody(body any) ([]byte,string,error) {
    if multipartBody,ok:=body.(*KajiMultipartBody);ok{return multipartBody.encode()}
    data,err:=json.Marshal(body)
    return data,"application/json",err
}

type kajiPartEncoding struct {ContentType string `json:"contentType"`;Headers map[string]struct{Required bool `json:"required"`} `json:"headers"`;Encoding map[string]kajiPartEncoding `json:"encoding"`;PrefixEncoding []kajiPartEncoding `json:"prefixEncoding"`;ItemEncoding *kajiPartEncoding `json:"itemEncoding"`}
type kajiMultipartDefinition struct {ContentType string `json:"content_type"`;Encoding map[string]kajiPartEncoding `json:"encoding"`;PrefixEncoding []kajiPartEncoding `json:"prefix_encoding"`;ItemEncoding *kajiPartEncoding `json:"item_encoding"`}
func kajiDeclaredMultipart(body *KajiMultipartBody,metadata string)(*KajiMultipartBody,error){
 var definition kajiMultipartDefinition;if err:=json.Unmarshal([]byte(metadata),&definition);err!=nil{return nil,err}
 return kajiPrepareMultipart(body,definition.ContentType,definition.Encoding,definition.PrefixEncoding,definition.ItemEncoding,0)
}
func kajiPrepareMultipart(body *KajiMultipartBody,contentType string,named map[string]kajiPartEncoding,prefix []kajiPartEncoding,item *kajiPartEncoding,depth int)(*KajiMultipartBody,error){
 if depth>16{return nil,fmt.Errorf("kaji: multipart nesting exceeds 16 levels")}
 if body==nil{return nil,fmt.Errorf("kaji: nested multipart body required")}
 result:=&KajiMultipartBody{ContentType:contentType,Parts:append([]KajiMultipartPart(nil),body.Parts...)}
 if body.ContentType!=""{result.ContentType=body.ContentType}
 for i:=range result.Parts{part:=&result.Parts[i];var encoding kajiPartEncoding
  if len(prefix)>0||item!=nil{if i<len(prefix){encoding=prefix[i]}else if item!=nil{encoding=*item}}else{encoding=named[part.Name]}
  if part.ContentType==""&&encoding.ContentType!=""{if strings.Contains(encoding.ContentType,","){return nil,fmt.Errorf("kaji: select explicit part content type from declared alternatives")};part.ContentType=encoding.ContentType}
  for name,header:=range encoding.Headers{if strings.EqualFold(name,"Content-Type"){continue};if header.Required&&len(part.Headers.Values(name))==0{return nil,fmt.Errorf("kaji: required multipart header %s missing",name)}}
  if part.Nested!=nil{nestedType:=part.ContentType;if nestedType==""{nestedType="multipart/mixed"};var err error;part.Nested,err=kajiPrepareMultipart(part.Nested,nestedType,encoding.Encoding,encoding.PrefixEncoding,encoding.ItemEncoding,depth+1);if err!=nil{return nil,err}}
 }
 return result,nil
}
