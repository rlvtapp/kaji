
// KajiMiddleware wraps the native HTTP execution boundary. Request context and
// response-body ownership follow net/http conventions. Wrappers may rewrite
// requests/responses, recover errors, or return a response without calling next.
// Middleware and its wrapped transport must support concurrent use.
type KajiMiddleware func(KajiHTTPClient) KajiHTTPClient

// KajiHTTPClientFunc adapts a function into an injectable transport.
type KajiHTTPClientFunc func(*http.Request) (*http.Response, error)
func (handler KajiHTTPClientFunc) Do(request *http.Request) (*http.Response, error) {
    return handler(request)
}
