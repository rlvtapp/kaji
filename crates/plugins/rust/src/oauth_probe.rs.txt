#[cfg(test)] mod native_oauth {
 use super::*;use crate::transport::{Transport,TransportFuture};use std::sync::atomic::{AtomicUsize,Ordering};
 struct Issuer { calls: AtomicUsize, lifetime:u64, bad:bool }
 impl Transport for Issuer {fn execute(&self,request:reqwest::Request)->TransportFuture<'_>{Box::pin(async move {let n=self.calls.fetch_add(1,Ordering::SeqCst)+1;assert_eq!(request.headers()["authorization"],"Basic JUMzJUJDJTNBaWQ6c2VjcmV0");assert!(std::str::from_utf8(request.body().unwrap().as_bytes().unwrap()).unwrap().contains("grant_type=client_credentials"));tokio::time::sleep(Duration::from_millis(20)).await;let body=if self.bad {"secret".to_owned()}else{serde_json::json!({"access_token":format!("t{n}"),"token_type":"Bearer","expires_in":self.lifetime}).to_string()};Ok(http::Response::builder().status(200).body(reqwest::Body::from(body)).unwrap().into())})}}
 struct Api {calls:AtomicUsize,always:bool}
 impl Transport for Api {fn execute(&self,request:reqwest::Request)->TransportFuture<'_>{let n=self.calls.fetch_add(1,Ordering::SeqCst)+1;let token=request.headers()["authorization"].to_str().unwrap().to_owned();let status=if self.always||n==1{401}else{200};Box::pin(async move{assert!(token.starts_with("Bearer t"));Ok(http::Response::builder().status(status).body(reqwest::Body::from("1")).unwrap().into())})}}
 #[tokio::test]async fn cache_singleflight_replay_and_cancellation(){
  let issuer=Arc::new(Issuer{calls:AtomicUsize::new(0),lifetime:3600,bad:false});let provider=Arc::new(OAuthClientCredentials::new("https://issuer.test/token","ü:id","secret").unwrap().with_refresh_leeway(Duration::ZERO).with_transport(issuer.clone()));
  let mut handles=Vec::new();for _ in 0..16{let p=provider.clone();handles.push(tokio::spawn(async move{p.token().await.unwrap()}));}for h in handles{assert_eq!(h.await.unwrap(),"t1")}assert_eq!(issuer.calls.load(Ordering::SeqCst),1);
  provider.invalidate("old");assert_eq!(provider.token().await.unwrap(),"t1");
  let api=Arc::new(Api{calls:AtomicUsize::new(0),always:false});let client=crate::Client::new("https://api.test").with_retry(crate::RetryConfig{max_attempts:1,initial_delay:Duration::ZERO,max_delay:Duration::ZERO}).with_token_provider(provider.clone()).with_transport(api.clone());assert!(client.get_thing().await.is_ok());assert_eq!(api.calls.load(Ordering::SeqCst),2);assert_eq!(issuer.calls.load(Ordering::SeqCst),2);
  let denied=Arc::new(Api{calls:AtomicUsize::new(0),always:true});let client=crate::Client::new("https://api.test").with_token_provider(provider.clone()).with_transport(denied.clone());assert!(client.get_thing().await.is_err());assert_eq!(denied.calls.load(Ordering::SeqCst),2);assert!(client.unsafe_thing().await.is_err());assert_eq!(denied.calls.load(Ordering::SeqCst),3);
  let zero=Arc::new(Issuer{calls:AtomicUsize::new(0),lifetime:0,bad:false});let p=OAuthClientCredentials::new("https://issuer.test/token","ü:id","secret").unwrap().with_transport(zero.clone());p.token().await.unwrap();p.token().await.unwrap();assert_eq!(zero.calls.load(Ordering::SeqCst),2);
  let bad=Arc::new(Issuer{calls:AtomicUsize::new(0),lifetime:3600,bad:true});let p=OAuthClientCredentials::new("https://issuer.test/token","ü:id","secret").unwrap().with_transport(bad);assert_eq!(p.token().await.unwrap_err().to_string(),"OAuth token acquisition failed");assert!(!format!("{p:?}").contains("secret"));
  let slow=Arc::new(Issuer{calls:AtomicUsize::new(0),lifetime:3600,bad:false});let p=Arc::new(OAuthClientCredentials::new("https://issuer.test/token","ü:id","secret").unwrap().with_transport(slow.clone()));let leader={let p=p.clone();tokio::spawn(async move{p.token().await})};tokio::time::sleep(Duration::from_millis(5)).await;leader.abort();assert!(leader.await.unwrap_err().is_cancelled());assert!(p.token().await.is_ok());assert_eq!(slow.calls.load(Ordering::SeqCst),2);
 }
}
#[cfg(test)] mod call_scopes {
 use super::*;use crate::transport::{Transport,TransportFuture};
 struct Driver(Mutex<Vec<(Option<String>,Option<Duration>)>>);
 impl Transport for Driver {fn execute(&self,request:reqwest::Request)->TransportFuture<'_>{self.0.lock().unwrap().push((request.headers().get("X-Scope").map(|x|x.to_str().unwrap().to_owned()),request.timeout().copied()));Box::pin(async{Ok(http::Response::builder().status(200).body(reqwest::Body::from("1")).unwrap().into())})}}
 #[tokio::test]async fn scopes_copy_headers_and_timeout_without_mutating_original(){let driver=Arc::new(Driver(Mutex::new(Vec::new())));let client=crate::Client::new("https://api.test").with_transport(driver.clone());let mut headers=reqwest::header::HeaderMap::new();headers.insert("X-Scope",reqwest::header::HeaderValue::from_static("scope"));let scoped=client.for_call(crate::CallOptions{headers,timeout:Some(Duration::from_millis(25))});assert!(scoped.get_thing().await.is_ok());assert!(client.get_thing().await.is_ok());assert_eq!(*driver.0.lock().unwrap(),vec![(Some("scope".into()),Some(Duration::from_millis(25))),(None,None)]);}
}
